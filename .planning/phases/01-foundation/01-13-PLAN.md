```yaml
- name: Resolve Vercel preview URL
  if: github.event_name == 'pull_request'
  id: vercel-preview
  run: |
    for i in {1..30}; do
      URL=$(gh pr view ${{ github.event.pull_request.number }} --json comments --jq '[.comments[] | select(.body | contains(".vercel.app")) | .body] | first' 2>/dev/null | grep -oE 'https://[a-z0-9-]+\.vercel\.app' | head -1)
      if [ -n "$URL" ]; then
        echo "preview-url=$URL" >> "$GITHUB_OUTPUT"
        echo "Resolved: $URL"
        break
      fi
      echo "Attempt $i: no preview URL yet, waiting 10s..."
      sleep 10
    done
    if [ -z "$URL" ]; then
      echo "::error::Timeout waiting for Vercel preview URL on PR"
      exit 1
    fi
  env:
    GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}

- name: Wait for preview to be ready
  if: github.event_name == 'pull_request'
  run: |
    PREVIEW_URL="${{ steps.vercel-preview.outputs.preview-url }}"
    for i in {1..30}; do
      if curl -fsS "$PREVIEW_URL/api/healthz" > /dev/null 2>&1; then break; fi
      sleep 10
    done

- name: Run Playwright smoke
  run: pnpm test:e2e
  env:
    PLAYWRIGHT_BASE_URL: ${{ steps.vercel-preview.outputs.preview-url }}
```
