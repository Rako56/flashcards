<!-- withErrorTracking wrapper -->
```ts
import { captureWithCorrelation } from './sentry'
import { logWithCorrelation } from './logger'

export function withErrorTracking<TArgs extends unknown[], TResult>(
  routeName: string,
  fn: (correlationId: string, ...args: TArgs) => Promise<TResult>
): (correlationId: string, ...args: TArgs) => Promise<TResult> {
  return async (correlationId, ...args) => {
    const log = logWithCorrelation(correlationId)
    try {
      return await fn(correlationId, ...args)
    } catch (err) {
      const eventId = captureWithCorrelation(err, correlationId, { route: routeName })
      log.error({ err, eventId }, `[${routeName}] caught exception`)
      throw err
    }
  }
}
```
